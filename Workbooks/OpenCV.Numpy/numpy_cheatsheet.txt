NUMPY CHEATSHEET
================

IMPORT
------
import numpy as np


ARRAY CREATION
--------------
np.array([1, 2, 3])              # from list
np.zeros((3, 4))                 # 3x4 array of 0s
np.ones((2, 3))                  # 2x3 array of 1s
np.full((2, 2), 7)               # 2x2 filled with 7
np.eye(3)                        # 3x3 identity matrix
np.arange(0, 10, 2)              # [0, 2, 4, 6, 8]
np.linspace(0, 1, 5)             # 5 evenly spaced points
np.random.rand(3, 3)             # uniform [0, 1)
np.random.randn(3, 3)            # standard normal
np.random.randint(0, 10, (2, 3)) # random ints


INSPECTING
----------
a.shape          # dimensions
a.ndim           # number of axes
a.size           # total elements
a.dtype          # data type
a.itemsize       # bytes per element
len(a)           # length of first axis


INDEXING & SLICING
------------------
a[0]             # first element / row
a[-1]            # last
a[1:4]           # slice
a[:, 0]          # first column (2D)
a[0, :]          # first row (2D)
a[1:3, 0:2]      # submatrix
a[a > 5]         # boolean mask
a[[0, 2, 4]]     # fancy indexing


RESHAPING
---------
a.reshape(3, 4)
a.flatten()                  # copy as 1D
a.ravel()                    # view as 1D
a.T                          # transpose
np.expand_dims(a, axis=0)    # add axis
np.squeeze(a)                # remove size-1 axes
a[:, np.newaxis]             # add column axis


COMBINING / SPLITTING
---------------------
np.concatenate([a, b], axis=0)
np.vstack([a, b])            # stack rows
np.hstack([a, b])            # stack columns
np.stack([a, b], axis=0)     # new axis
np.split(a, 3, axis=0)
np.tile(a, (2, 3))
np.repeat(a, 3)


MATH (ELEMENT-WISE)
-------------------
a + b, a - b, a * b, a / b
a ** 2, np.sqrt(a)
np.exp(a), np.log(a)
np.sin(a), np.cos(a)
np.abs(a), np.round(a)
np.clip(a, 0, 1)


LINEAR ALGEBRA
--------------
a @ b                        # matrix multiply
np.dot(a, b)
np.linalg.inv(a)             # inverse
np.linalg.det(a)             # determinant
np.linalg.eig(a)             # eigenvalues, vectors
np.linalg.solve(A, b)        # solve Ax = b
np.linalg.norm(a)


AGGREGATION
-----------
a.sum(), a.mean(), a.std(), a.var()
a.min(), a.max()
a.argmin(), a.argmax()
a.sum(axis=0)                # column sums
a.sum(axis=1)                # row sums
np.median(a), np.percentile(a, 50)
np.cumsum(a), np.cumprod(a)


COMPARISON / LOGIC
------------------
a == b, a > b
np.array_equal(a, b)
np.all(a > 0), np.any(a > 0)
np.where(a > 0, 1, -1)       # ternary
np.isnan(a), np.isinf(a)


SORTING & UNIQUE
----------------
np.sort(a)
a.argsort()                  # indices
np.unique(a)
np.unique(a, return_counts=True)


I/O
---
np.save('file.npy', a)
np.load('file.npy')
np.savetxt('file.csv', a, delimiter=',')
np.loadtxt('file.csv', delimiter=',')


BROADCASTING RULES
------------------
1. Align shapes from the right.
2. Dimensions must match or be 1.
3. Size-1 dims are stretched.

Example:
  (3, 4) + (4,)    -> (3, 4)
  (3, 1) + (1, 4)  -> (3, 4)


COPY VS VIEW
------------
b = a            # reference
b = a.view()     # shallow (shares data)
b = a.copy()     # independent copy
